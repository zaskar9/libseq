mode oracle
#! input SYNC.TXT
#! secs 40
#! irq
load TTSSEQ.DLL
call.w SEQOPEN 0000 0000
buf 0 16
call.w SEQLOADCAKEWALKFILE "C:\PROBE\ORC\QUANT.WRK" 0000 null 0000 0000
echo == clock source values
call.w SEQSETSRCCLOCK 0001
call.w SEQGETSRCCLOCK
call.w SEQSETSRCCLOCK 0002
call.w SEQGETSRCCLOCK
call.w SEQSETSRCCLOCK 0003
call.w SEQGETSRCCLOCK
call.w SEQSETSRCCLOCK 0004
call.w SEQGETSRCCLOCK
call.w SEQSETSRCCLOCK 0103
call.w SEQGETSRCCLOCK
call.w SEQSETSRCCLOCK ffff
call.w SEQGETSRCCLOCK
call.w SEQSETSRCCLOCK 0000
call.w SEQGETSRCCLOCK
echo == MIDI sync: wait for Start (clocks at 1500 to 1600 do not end it)
call.w SEQSETSRCCLOCK 0001
call.w SEQGETSRCCLOCK
call.w SEQSETPLAYINGSEQ 0001
call.w SEQSETSTANDBYSEQ 0000
call.w SEQEXPIRED
call.w SEQPLAYSTART null
call.d SEQGETNOW 0001
echo == Start at 2000, two clocks: ticks 0 to 9 are played; Now is the time of the last event
echo == played, else 5 ticks more than before
idle 400
call.d SEQGETNOW 0001
echo == six more clocks
idle 500
call.d SEQGETNOW 0001
echo == 28 clocks at twice the rate
idle 1000
call.d SEQGETNOW 0001
call.w SEQEXPIRED
echo == Stop at 4000
idle 300
call.d SEQGETNOW 0001
call.w SEQEXPIRED
echo == Continue at 4500 without a Song Position: from where it stopped; four clocks; Stop
call.w SEQPLAYSTART null
call.d SEQGETNOW 0001
idle 450
call.d SEQGETNOW 0001
idle 300
call.d SEQGETNOW 0001
echo == Song Position 480 at 5100 while stopped; Continue at 6200
call.w SEQPLAYSTART null
call.d SEQGETNOW 0001
echo == Song Position 0 at 6500 while playing; Stop at 6800
idle 400
call.d SEQGETNOW 0001
idle 400
call.d SEQGETNOW 0001
echo == Continue at 7800: from the Song Position kept; Stop at 8100
call.w SEQPLAYSTART null
call.d SEQGETNOW 0001
idle 400
call.d SEQGETNOW 0001
echo == Start at 8600 while the start callback polls; four clocks; Stop
cbret 0
fill 0 0 ee ee ee ee
call.w SEQPLAYSTART cb3
call.d SEQGETNOW 0001
idle 450
call.d SEQGETNOW 0001
idle 250
echo == Stop at 9400 while waiting is kept: the Start at 9600 stops at once
call.w SEQPLAYSTART null
call.d SEQGETNOW 0001
idle 300
call.d SEQGETNOW 0001
call.w SEQEXPIRED
echo == MIDI Time Code from 00:00:10:00 at 10300 (ENG_ORDR, which ends before 00:00:10:00):
echo == the start returns 2 s after the code locks, at the time of the code
call.w SEQLOADCAKEWALKFILE "C:\PROBE\ORC\ENG_ORDR.WRK" 0000 null 0000 0000
call.w SEQSETSRCCLOCK 0003
call.w SEQGETSRCCLOCK
call.w SEQSETPLAYINGSEQ 0002
fill 0 0 ee ee ee ee
call.w SEQPLAYSTART null
call.d SEQGETNOW 0002
echo == the code stops at 13800; the engine runs on for 0.5 s, then stops
idle 2500
call.w SEQEXPIRED
echo == the callback gives up at once
cbret 1
fill 0 0 ee ee ee ee
call.w SEQPLAYSTART cb3
call.w SEQEXPIRED
call.w SEQCLOSE
