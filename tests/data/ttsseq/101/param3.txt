mode oracle
#! secs 40
load TTSSEQ.DLL
call.w SEQOPEN 0000 0000
buf 1 16
call.w SEQCREATESEQ 0000 0000
echo == created
call.d SEQGETSEQPARAM 0001 0003
call.w SEQDESTROYSEQ 0001
call.w SEQLOADCAKEWALKFILE "C:\PROBE\ORC\ENG_ORDR.WRK" 0000 null 0000 0000
echo == loaded
call.d SEQGETSEQPARAM 0001 0003
call.w SEQSETSEQPARAM 0001 0003 d:00000000
echo == cleared
call.d SEQGETSEQPARAM 0001 0003
call.w SEQSETSEQPARAM 0001 0003 d:00000000
echo == set to 5
call.w SEQSETSEQPARAM 0001 0003 d:00000005
call.d SEQGETSEQPARAM 0001 0003
call.w SEQSETSEQPARAM 0001 0003 d:00000000
echo == another parameter
call.w SEQSETSEQPARAM 0001 0002 d:00000001
call.d SEQGETSEQPARAM 0001 0003
call.w SEQSETSEQPARAM 0001 0003 d:00000000
echo == insert, put, delete and move events
call.d SEQSEEK 0100 0000 d:00000000
fill 1 0 78 00 00 b0 07 64 00 00
call.w SEQINSEVT 0100 @1
call.d SEQSEEK 0100 0000 d:00000002
call.w SEQPUTEVT 0100 @1
call.w SEQDELEVT 0100
call.d SEQSEEK 0100 0000 d:00000001
call.d SEQMOVEEVT 0100 d:00000300
call.d SEQGETSEQPARAM 0001 0003
call.w SEQSETSEQPARAM 0001 0003 d:00000000
echo == insert a tempo
call.w SEQINSTEMPO 0001 d:000001e0 2ee0
call.d SEQGETSEQPARAM 0001 0003
call.w SEQSETSEQPARAM 0001 0003 d:00000000
echo == clone and destroy a stream
call.w SEQCLONESTREAM 0001 0100 0000
let 2
call.w SEQDESTROYSTREAM %2
call.d SEQGETSEQPARAM 0001 0003
call.w SEQSETSEQPARAM 0001 0003 d:00000000
echo == save
call.w SEQSAVECAKEWALKFILE 0001 "C:\PROBE\ORC\OUT\P3.WRK" d:00000000 null d:00000000
call.d SEQGETSEQPARAM 0001 0003
call.w SEQSETSEQPARAM 0001 0003 d:00000000
echo == set Now
call.d SEQSETNOW 0001 d:000000f0
call.d SEQGETSEQPARAM 0001 0003
call.w SEQSETSEQPARAM 0001 0003 d:00000000
echo == set Now to the same value
call.d SEQSETNOW 0001 d:000000f0
call.d SEQGETSEQPARAM 0001 0003
call.w SEQSETSEQPARAM 0001 0003 d:00000000
echo == rewind
call.d SEQREWIND 0001
call.d SEQGETSEQPARAM 0001 0003
call.w SEQSETSEQPARAM 0001 0003 d:00000000
echo == make it the playing sequence
call.w SEQSETPLAYINGSEQ 0001
call.w SEQSETSTANDBYSEQ 0000
call.d SEQGETSEQPARAM 0001 0003
call.w SEQSETSEQPARAM 0001 0003 d:00000000
echo == start
call.w SEQPLAYSTART null
call.d SEQGETSEQPARAM 0001 0003
call.w SEQSETSEQPARAM 0001 0003 d:00000000
echo == play on, then stop
idle 600
call.d SEQGETSEQPARAM 0001 0003
call.w SEQPLAYSTOP
idle 200
call.d SEQGETSEQPARAM 0001 0003
call.w SEQSETSEQPARAM 0001 0003 d:00000000
echo == record start (no count-in)
call.w SEQSETSEQPARAM 0001 0010 d:00000000
cbret 0
call.w SEQRECORDSTART 0000 0000 0000 0000 0000 cb3
call.d SEQGETSEQPARAM 0001 0003
call.w SEQSETSEQPARAM 0001 0003 d:00000000
echo == record stop
idle 300
call.w SEQRECORDSTOP
call.d SEQGETSEQPARAM 0001 0003
echo == E3_SHTL: play, clear after the start, read before and after the wrap (960 to 480)
call.w SEQDESTROYSEQ 0001
call.w SEQLOADCAKEWALKFILE "C:\PROBE\ORC\E3_SHTL.WRK" 0000 null 0000 0000
call.w SEQSETPLAYINGSEQ 0001
call.w SEQPLAYSTART null
idle 200
call.w SEQSETSEQPARAM 0001 0003 d:00000000
idle 3000
call.d SEQGETSEQPARAM 0001 0003
idle 1500
call.d SEQGETSEQPARAM 0001 0003
call.w SEQPLAYSTOP
call.w SEQCLOSE
