mode oracle
# SeqCloneStream: the flag copies the parameters; the events are copied either way.
load TTSSEQ.DLL
call.w SEQWIRE 0001
call.w SEQOPEN 0000 0000
call.w SEQLOADCAKEWALKFILE "C:\PROBE\ORC\ENG_ORDR.WRK" 0000 null 0000 0000
call.w SEQSETSTREAMPARAM 0100 0005 d:0000000a
call.w SEQSETSTREAMPARAM 0100 0006 d:00000064
call.w SEQSETSTREAMPARAM 0100 000b d:00000003
echo == clone 0000
call.w SEQCLONESTREAM 0001 0100 0000
let 3
call.d SEQSTREAMLEN %3
call.d SEQGETSTREAMPARAM %3 0000
call.d SEQGETSTREAMPARAM %3 0001
call.d SEQGETSTREAMPARAM %3 0002
call.d SEQGETSTREAMPARAM %3 0003
call.d SEQGETSTREAMPARAM %3 0004
call.d SEQGETSTREAMPARAM %3 0005
call.d SEQGETSTREAMPARAM %3 0006
call.d SEQGETSTREAMPARAM %3 0007
call.d SEQGETSTREAMPARAM %3 0008
call.d SEQGETSTREAMPARAM %3 0009
call.d SEQGETSTREAMPARAM %3 000a
call.d SEQGETSTREAMPARAM %3 000b
call.d SEQGETSTREAMPARAM %3 000c
call.d SEQGETSTREAMPARAM %3 000d
call.d SEQGETSTREAMPARAM %3 000e
call.d SEQGETSTREAMPARAM %3 000f
echo == clone 0001
call.w SEQCLONESTREAM 0001 0100 0001
let 3
call.d SEQSTREAMLEN %3
call.d SEQGETSTREAMPARAM %3 0000
call.d SEQGETSTREAMPARAM %3 0001
call.d SEQGETSTREAMPARAM %3 0002
call.d SEQGETSTREAMPARAM %3 0003
call.d SEQGETSTREAMPARAM %3 0004
call.d SEQGETSTREAMPARAM %3 0005
call.d SEQGETSTREAMPARAM %3 0006
call.d SEQGETSTREAMPARAM %3 0007
call.d SEQGETSTREAMPARAM %3 0008
call.d SEQGETSTREAMPARAM %3 0009
call.d SEQGETSTREAMPARAM %3 000a
call.d SEQGETSTREAMPARAM %3 000b
call.d SEQGETSTREAMPARAM %3 000c
call.d SEQGETSTREAMPARAM %3 000d
call.d SEQGETSTREAMPARAM %3 000e
call.d SEQGETSTREAMPARAM %3 000f
echo == source
call.d SEQGETSTREAMPARAM 0100 0005
call.d SEQGETSTREAMPARAM 0100 0006
call.d SEQGETSTREAMPARAM 0100 000b
call.w SEQCLOSE
